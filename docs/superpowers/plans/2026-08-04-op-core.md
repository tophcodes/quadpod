```rust
pub fn load_or_generate(path: &Path) -> Result<Self, KeyError> {
    let text = match std::fs::read_to_string(path) {
        Ok(t) => t,
        Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
            let mut jwk = josekit::jwk::Jwk::generate_ec_key(josekit::jwk::alg::ec::EcCurve::P256)
                .map_err(|e| KeyError::Malformed { path: path.into(), reason: e.to_string() })?;
            jwk.set_algorithm("ES256");
            jwk.set_key_id(thumbprint(&jwk));
            let body = serde_json::json!({ "keys": [&jwk] }).to_string();
            use std::io::Write;
            use std::os::unix::fs::OpenOptionsExt;
            let mut f = std::fs::OpenOptions::new()
                .write(true).create_new(true).mode(0o600).open(path)
                .map_err(|e| KeyError::Io { path: path.into(), source: e })?;
            f.write_all(body.as_bytes())
                .map_err(|e| KeyError::Io { path: path.into(), source: e })?;
            return Ok(Self { keys: vec![jwk] });
        }
        Err(e) => return Err(KeyError::Io { path: path.into(), source: e }),
    };
    let parsed: serde_json::Value = serde_json::from_str(&text)
        .map_err(|e| KeyError::Malformed { path: path.into(), reason: e.to_string() })?;
    let raw = parsed.get("keys").and_then(|k| k.as_array())
        .ok_or_else(|| KeyError::Malformed { path: path.into(), reason: "no `keys` array".into() })?;
    let mut keys = Vec::new();
    for v in raw {
        let map = v.as_object()
            .ok_or_else(|| KeyError::Malformed { path: path.into(), reason: "key is not an object".into() })?;
        let mut jwk = josekit::jwk::Jwk::from_map(map.clone())
            .map_err(|e| KeyError::Malformed { path: path.into(), reason: e.to_string() })?;
        if jwk.key_id().is_none() {
            let t = thumbprint(&jwk);
            jwk.set_key_id(t);
        }
        keys.push(jwk);
    }
    if keys.is_empty() {
        return Err(KeyError::Empty { path: path.into() });
    }
    Ok(Self { keys })
}
```

Plus a private `fn thumbprint(jwk: &Jwk) -> String` in the same file: RFC 7638 — build the canonical member set for the key type (`crv`,`kty`,`x`,`y` for EC; `e`,`kty`,`n` for RSA) as a JSON object with sorted keys, SHA-256, base64url-no-pad. `src/auth/testsupport.rs::TestClient::jkt` and the binding check in `src/auth/dpop.rs` already compute exactly this — copy that construction (same crates: `sha2`/`base64` as used there; check those files for the exact imports rather than adding new dependencies). Delete the placeholder `let _ = …` lines from the skeleton body.

```rust
pub fn public_jwks(&self) -> serde_json::Value {
    let keys: Vec<serde_json::Value> = self
        .keys
        .iter()
        .map(|k| {
            let public = k.to_public_key().expect("every loaded key has a public half");
            serde_json::to_value(&public).expect("a Jwk serializes")
        })
        .collect();
    serde_json::json!({ "keys": keys })
}

pub fn signing_algs(&self) -> Vec<String> {
    let mut out: Vec<String> = Vec::new();
    for k in &self.keys {
        if let Some(alg) = k.algorithm() {
            if !out.iter().any(|a| a == alg) {
                out.push(alg.to_string());
            }
        }
    }
    out
}
```

```rust
pub(crate) fn sign_jwt(&self, payload: &JwtPayload) -> String {
    let key = &self.keys[0];
    let mut header = josekit::jws::JwsHeader::new();
    header.set_token_type("JWT");
    header.set_key_id(key.key_id().expect("every loaded key has a kid"));
    let signer: Box<dyn josekit::jws::JwsSigner> = match key.algorithm() {
        Some("RS256") => Box::new(josekit::jws::RS256.signer_from_jwk(key).expect("RS256 signer")),
        _ => Box::new(josekit::jws::ES256.signer_from_jwk(key).expect("ES256 signer")),
    };
    josekit::jwt::encode_with_signer(payload, &header, &*signer).expect("sign JWT")
}
```

```rust
pub fn mint_access_token(
    keys: &KeySet,
    space: &StorageSpace,
    webid: &NamedNode,
    jkt: &str,
    now_unix: i64,
) -> String {
    let issuer = space.root().graph_iri().to_string();
    let mut payload = josekit::jwt::JwtPayload::new();
    payload.set_issuer(&issuer);
    payload.set_subject(webid.as_str());
    payload.set_audience(vec!["solid"]);
    payload
        .set_claim("webid", Some(serde_json::Value::String(webid.as_str().to_string())))
        .expect("webid claim");
    payload
        .set_claim("cnf", Some(serde_json::json!({ "jkt": jkt })))
        .expect("cnf claim");
    payload
        .set_claim("jti", Some(serde_json::Value::String(uuid::Uuid::new_v4().to_string())))
        .expect("jti claim");
    payload
        .set_claim("iat", Some(serde_json::Value::from(now_unix)))
        .expect("iat claim");
    payload
        .set_claim("exp", Some(serde_json::Value::from(now_unix + ACCESS_TOKEN_TTL_SECS)))
        .expect("exp claim");
    keys.sign_jwt(&payload)
}
```

```rust
    pub fn op_keys(&self) -> Result<Option<crate::op::KeySet>, String> {
        let Some(path) = &self.op_signing_keys else {
            return Ok(None);
        };
        let root = self.base_uri.root().graph_iri().to_string();
        let after_scheme = root
            .split_once("://")
            .map(|(_, rest)| rest)
            .unwrap_or(root.as_str());

            ));
        }
        crate::op::KeySet::load_or_generate(path)
            .map(Some)
            .map_err(|e| format!("--op-signing-keys: {e}"))
    }
```

```rust
pub(super) async fn fixture_with_op() -> (Fixture, Arc<crate::op::KeySet>, std::path::PathBuf) {
    let path = std::env::temp_dir().join(format!("op-fixture-{}.json", uuid::Uuid::new_v4()));

        .collect();
    let mut issuers = StaticWebIdIssuers::new();

        replay: f.replay.clone(),

    };
```

```rust
async fn handle_well_known(
    State(state): State<AppState>,
    rest: Option<Path<String>>,
) -> Response {
    let Some(keys) = &state.op_keys else {
        return StatusCode::NOT_FOUND.into_response();
    };
    match rest.as_deref() {
        Some("openid-configuration") => {
            axum::Json(crate::op::discovery::document(&state.space, keys)).into_response()
        }
        Some("jwks.json") => (
            [(header::CONTENT_TYPE, "application/jwk-set+json")],
            crate::op::KeySet::public_jwks(keys).to_string(),
        )
            .into_response(),
        _ => StatusCode::NOT_FOUND.into_response(),
    }
}
```

Notes: `Option<Path<String>>` needs `as_deref()` through both layers — if the double deref does not compile, match on `rest.map(|Path(s)| s)` first. `axum::Json` sets `application/json`. No guard call: the two documents are public by design, and they read no store.
